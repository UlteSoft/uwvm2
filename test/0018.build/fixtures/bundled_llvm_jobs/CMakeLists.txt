cmake_minimum_required(VERSION 3.20)
project(ForwardedJobs NONE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(kind llvm runtime)
 set(outputs)
 foreach(i RANGE 1 8)
  set(output "${CMAKE_BINARY_DIR}/done/${kind}/${i}")
  add_custom_command(OUTPUT "${output}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/worker.py" "${CMAKE_BINARY_DIR}/done/${kind}.json" "${output}" VERBATIM)
  list(APPEND outputs "${output}")
 endforeach()
 if(kind STREQUAL "llvm")
  add_custom_target(llvm-libraries DEPENDS ${outputs})
 else()
  add_custom_target(cxx_static DEPENDS ${outputs})
  add_custom_target(cxxabi_static DEPENDS cxx_static)
  add_custom_target(unwind_static DEPENDS cxx_static)
 endif()
endforeach()
